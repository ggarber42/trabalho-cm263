Selecão de variáveis

Código 
# filtrar as variáveis de interesse
dados<-dados |> 
  dplyr::select(idsolicitacao,situacaosolicitacao,data_solicitacao,corregulador,
  diasfilaespera,complexidade,especialidademae,cid,idpaciente,idade,cnes_solicitante,situacaoagenda,
  datahora_consulta,cnesexecutante, possuiunidadeindicada, favela_comunidade,taxa_alf_geral,
  coordenadoria_saude,distrito_sanitario, renda_media_winsor) |> 
    filter(complexidade=="MEDIA") |>
    rename(censo_favela_comunidade = favela_comunidade,
           censo_taxa_alfabetizacao = taxa_alf_geral,
           censo_coordenadoria_saude = coordenadoria_saude,
           censo_distrito_sanitario = distrito_sanitario,
           censo_renda_media_winsor = renda_media_winsor)
Código 
# Transformações de variáveis
datas<- c("data_solicitacao","datahora_consulta")

categoricas<- c("situacaosolicitacao","corregulador","complexidade","especialidademae",
              "cid","cnes_solicitante","situacaoagenda","cnesexecutante",
              "possuiunidadeindicada","censo_coordenadoria_saude","censo_distrito_sanitario")

numericas<- c("diasfilaespera","idade","censo_taxa_alfabetizacao","censo_renda_media_winsor","censo_favela_comunidade")

dados<-dados |> 
  mutate(across(all_of(datas), ymd_hms)) |>
  mutate(across(all_of(categoricas), as.factor)) |>
  mutate(across(all_of(numericas), as.numeric))
Código 
# Filtrar a variavel complexidade para "MEDIA"
dados<-dados |>
  filter(complexidade=="MEDIA") |> 
  dplyr::select(-complexidade)
